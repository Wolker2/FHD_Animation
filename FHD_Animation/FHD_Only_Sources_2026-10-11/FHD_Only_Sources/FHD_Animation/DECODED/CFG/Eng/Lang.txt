ShipInfo ^{
    AddInfo ^{
        CustomInfos ^{
            FHD_StationEmblem_Info ~{
                Description=NoShow
                Icon=0
                Name=
                OnActCodeTypes=t_OnShowingShipInfo
                OnActCode ~{
                    01=dword station = ScriptItemActShip();
                    02=if(!station) station = CurShip;
                    03=if(!station) exit;
                    04=int owner = ShipOwner(station);
                    05=if(owner == Fei)
                    06={
                    07=    if(CurrentForm() == 'StarMap') InterfaceImage('StarMap','InfoShipEmRace','Alpha,Bm.FHDStationEmblem.Fei');
                    08=    else if(CurrentForm() == 'Hangar') InterfaceImage('Hangar','InfoShipEmRace','Alpha,Bm.FHDStationEmblem.Fei');
                    09=}
                    10=else if(owner == Peleng)
                    11={
                    12=    if(CurrentForm() == 'StarMap') InterfaceImage('StarMap','InfoShipEmRace','Alpha,Bm.FHDStationEmblem.Peleng');
                    13=    else if(CurrentForm() == 'Hangar') InterfaceImage('Hangar','InfoShipEmRace','Alpha,Bm.FHDStationEmblem.Peleng');
                    14=}
                }
            }
        }
    }
}
